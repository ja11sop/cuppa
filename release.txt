# Setup virtualenv if not exist
cd ..
python -m virtualenv venv
source venv/bin/activate
pip install -r requirements.txt
cd cuppa/

# Lint
flake8 cuppa
pylint -E cuppa

# Tests
pytest -m unit
pytest -m integration

# Release notes
# The changelog section is already open and filled in (scripts.start_release ran with the
# first commit of the cycle and entries were written as the work landed). Closing it dates
# the section, drops the .dev from cuppa/VERSION, prunes empty subsections, and closes the
# compare link:
python -m scripts.finish_release

# Commit, then tag. The release workflow re-runs scripts.check_release on the tag.
git commit -am "Release $(cat cuppa/VERSION)"
git tag -a "v$(cat cuppa/VERSION)" -m "cuppa $(cat cuppa/VERSION)"
git push origin master "v$(cat cuppa/VERSION)"

# Create the GitHub release from the CHANGELOG.md section for that version

# Build
rm -rf dist/ build/ && python setup.py sdist bdist_wheel

# Upload to PyPi
twine upload dist/*
