# Setup virtualenv if not exist
cd ..
python -m virtualenv venv
source venv/bin/activate
pip install -r requirements.txt
cd cuppa/

# Lint
flake8 cuppa
pylint -E cuppa

# Tests
pytest -m unit
pytest -m integration

# Build
rm -rf dist/ build/ && python setup.py sdist bdist_wheel

# Upload to PyPi
twine upload dist/*
